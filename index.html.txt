  <script>
    // CẤU HÌNH WEB APP URL CỦA GOOGLE SHEET Ở ĐÂY (NẾU CÓ)
    const GOOGLE_SHEET_API = ""; 

    // Dữ liệu mẫu ban đầu
    let staffData = [
      { name: "Hieunt310", code: "hieunt310", exp: "13 - 24T", target: 50 },
      { name: "Quoclv", code: "quoclv", exp: ">24T", target: 50 },
      { name: "ThuTU", code: "thutu", exp: "1 - 6T", target: 20 },
      { name: "Luannt68", code: "luannt68", exp: ">24T", target: 35 }
    ];

    let isAdmin = true;

    function renderStaff() {
      const tbody = document.getElementById('staffTableBody');
      tbody.innerHTML = '';
      staffData.forEach((s, idx) => {
        tbody.innerHTML += `
          <tr>
            <td class="p-3"><input type="text" class="border rounded px-2 py-1 w-full text-sm" value="${s.name}"></td>
            <td class="p-3"><input type="text" class="border rounded px-2 py-1 w-full text-sm" value="${s.code}"></td>
            <td class="p-3">
              <select class="border rounded px-2 py-1 text-sm">
                <option ${s.exp === '1 - 6T' ? 'selected' : ''}>1 - 6T</option>
                <option ${s.exp === '7 - 12T' ? 'selected' : ''}>7 - 12T</option>
                <option ${s.exp === '13 - 24T' ? 'selected' : ''}>13 - 24T</option>
                <option ${s.exp === '>24T' ? 'selected' : ''}>>24T</option>
              </select>
            </td>
            <td class="p-3"><input type="number" class="border rounded px-2 py-1 w-24 text-sm font-semibold text-blue-900" value="${s.target}"></td>
            <td class="p-3 text-center">
              <button onclick="removeRow(${idx})" class="text-rose-600 hover:text-rose-800 text-xs font-semibold">Xóa</button>
            </td>
          </tr>
        `;
      });
      document.getElementById('statStaff').innerText = staffData.length;
      document.getElementById('statTotalHD').innerText = staffData.reduce((a, b) => a + Number(b.target), 0);
      document.getElementById('statRate').innerText = "84.5%";
    }

    function addRow() {
      staffData.push({ name: "NV Mới", code: "account", exp: "1 - 6T", target: 20 });
      renderStaff();
    }

    function removeRow(idx) {
      staffData.splice(idx, 1);
      renderStaff();
    }

    function switchTab(tabName) {
      ['report', 'ranking', 'import'].forEach(t => {
        document.getElementById('content-' + t).classList.add('hidden');
        document.getElementById('tab-' + t).classList.remove('tab-active');
        document.getElementById('tab-' + t).classList.add('text-slate-600');
      });
      document.getElementById('content-' + tabName).classList.remove('hidden');
      document.getElementById('tab-' + tabName).classList.add('tab-active');
      document.getElementById('tab-' + tabName).classList.remove('text-slate-600');
    }

    function toggleRole() {
      isAdmin = !isAdmin;
      document.getElementById('roleBtn').innerText = isAdmin ? 'Quyền: Admin (Bấm đổi)' : 'Quyền: Nhân viên (Bấm đổi)';
    }

    // Khởi chạy ban đầu
    renderStaff();
  </script>
